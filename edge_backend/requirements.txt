# ONNX Runtime. `onnxruntime` and `onnxruntime-gpu` install the SAME import
# package, so they must never both be present: whichever was installed last
# silently wins. scripts/bootstrap.py removes the loser and reinstalls the
# winner.
# - x86_64 Linux/Windows: the GPU build. It also contains the CPU provider, so
#   it runs (on CPU) on a machine without an NVIDIA GPU. The [cuda,cudnn]
#   extras pull the nvidia-* CUDA/cuDNN pip wheels, so no system CUDA toolkit
#   is needed, only the NVIDIA driver.
# - everything else (aarch64 boards, macOS): the plain build, which carries
#   the CPU provider (and CoreML on macOS).
# Keep these markers in sync with ort_default_flavour() in scripts/bootstrap.py.
onnxruntime-gpu[cuda,cudnn]>=1.30 ; (sys_platform == "linux" or sys_platform == "win32") and (platform_machine == "x86_64" or platform_machine == "AMD64")
# (PEP 508 has no `not`, so the second marker is the first one negated by hand.)
onnxruntime>=1.30 ; (sys_platform != "linux" and sys_platform != "win32") or (platform_machine != "x86_64" and platform_machine != "AMD64")
